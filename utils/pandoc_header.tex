% ---------------------------------------------------------------------------
% LaTeX header injected into every pandoc Markdown -> PDF conversion
% (see utils/report_utils.py::convert_md_to_pdf).
%
% 1) longtable page-split fix
% ---------------------------
% pandoc renders every multi-row Markdown table as a `longtable`.  When such a
% table has to be broken across a page, longtable's own output routine
% (`\LT@output`) builds the partial column with
%
%     \setbox\@cclv\vbox{\unvbox\@cclv\copy\LT@foot\vss}
%
% The `\vss` is glue with *infinite shrinkability*.  The current LaTeX output
% routine subsequently \vsplit's that box, and TeX then raises
%
%     ! Infinite glue shrinkage found in box being split.
%
% TeX recovers on its own ("offensive shrinkability has been made finite"),
% but pandoc always invokes the engine with `-halt-on-error`, so the run is
% aborted and no PDF is ever written.  Replacing the infinite *shrink* (`\vss`)
% with infinite *stretch* (`\vfil`) keeps longtable's intent -- push the
% repeated foot to the bottom of the partial column -- while producing a box
% that is legal to split.
%
% This is engine independent (pdflatex/xelatex/lualatex all hit it) and depends
% on nothing in the report content, so it fixes present and future reports
% alike.  Both occurrences of `\vss` in `\LT@output` are patched: the first
% covers `\end{longtable}` when the last foot no longer fits, the second the
% ordinary mid-table page break.
% ---------------------------------------------------------------------------
\makeatletter
\@ifundefined{LT@output}{}{%
  \RequirePackage{etoolbox}%
  \patchcmd{\LT@output}{\vss}{\vfil}{}{}%
  \patchcmd{\LT@output}{\vss}{\vfil}{}{}%
}
\makeatother

% ---------------------------------------------------------------------------
% 3) Wide-table safety net
% ---------------------------------------------------------------------------
% Some report tables (e.g. the pairwise-difference matrices) carry long model
% identifiers that cannot be hyphenated.  These only produce "Overfull \hbox"
% warnings, but shrinking the table font and allowing extra stretch keeps them
% inside the text block.
\setlength{\emergencystretch}{3em}
\setlength{\tabcolsep}{4pt}
\makeatletter
\@ifundefined{LTleft}{}{\setlength{\LTleft}{0pt}\setlength{\LTright}{0pt}}
\makeatother
